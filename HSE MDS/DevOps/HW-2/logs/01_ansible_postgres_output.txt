- name: Initialize PostgreSQL cluster
  ansible.builtin.command:
    cmd: "/usr/lib/postgresql/{{ postgres__version }}/bin/pg_ctl initdb -D {{ postgres__data_dir }}"
    creates: "{{ postgres__data_dir }}/PG_VERSION"
  become: true
  become_user: postgres

- name: Install PostgreSQL systemd service
  ansible.builtin.template:
    src: postgresql-custom.service.j2
    dest: /etc/systemd/system/postgresql-custom.service
    owner: root
    group: root
    mode: '0644'

- name: Start PostgreSQL service
  ansible.builtin.systemd_service:
    name: postgresql-custom
    state: started
    enabled: true
    daemon_reload: true

- name: Configure PostgreSQL settings
  ansible.builtin.lineinfile:
    path: "{{ postgres__data_dir }}/postgresql.conf"
    regexp: '^\s*#?\s*{{ item.key }}\s*='
    line: "{{ item.key }} = {{ item.value }}"
  loop: "{{ postgres__config | dict2items }}"
  notify: Restart PostgreSQL
